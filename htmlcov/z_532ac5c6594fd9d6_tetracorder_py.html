1""" 

2Decoder for the tetracorder "expert system" command file (``cmd.lib.setup.*``). 

3 

4Tetracorder's expert system file is a plain-text description of every material it 

5maps: which library spectrum identifies it, the spectral features (continuum 

6endpoints) that define it, and the fit/depth constraints applied. 

7 

8The main class :class:`TetraDecoder` parses these expert system files into structured 

9dictionaries and provides methods to export the data to various formats (CSV, DataFrame). 

10It automatically resolves variable references from companion files (``cmd.lib.setup.variables``, 

11``cmds.start.*``) when they are present. 

12 

13Examples 

14-------- 

15Basic usage to decode an expert system file: 

16 

17>>> decoder = TetraDecoder("cmd.lib.setup.t5.51d") 

18>>> decoder.blocks # List of decoded material records 

19>>> df = decoder.table() # Get as pandas DataFrame 

20 

21Decode specific groups and export to CSV: 

22 

23>>> decoder = TetraDecoder("cmd.lib.setup.t5.51d", groups=(1, 2)) 

24>>> decoder.export_csv("output.csv", reference="reference_matrix.csv") 

25 

26Access decoded records programmatically: 

27 

28>>> for block in decoder.blocks: 

29... print(block["name"], block["library"], block["record"]) 

30"""

42class TetraDecoder: 

43 """ 

44 Decode a Tetracorder expert system file into structured per-material records. 

45 

46 The expert system file (``cmd.lib.setup.<version>``) is organized into "group 

47 blocks", one per mapped material. Each block names the library spectrum used to 

48 identify the material, the spectral features (continuum endpoints plus fit/depth 

49 thresholds) that define it, the constituent constraints applied, and the output 

50 file the mapper writes. :meth:`decode` walks the file and turns every block whose 

51 ``group`` is in ``only`` into a plain ``dict`` (see :meth:`parse_block` for the 

52 schema); the results are stored on :attr:`blocks`. 

53 

54 Several fields in the file are written as ``[NAME]`` references to variables 

55 defined in the companion ``cmd.lib.setup.variables`` file, and output paths are 

56 keyed by group directory in ``cmds.start.*``. When those sidecar files are found 

57 next to the expert system file they are parsed up front (:attr:`vars`, 

58 :attr:`groups`) and substituted in during decoding. 

59 

60 Attributes 

61 ---------- 

62 file : Path 

63 Resolved path to the expert system file being decoded. 

64 vars : dict[str, str] or None 

65 Variable name -> value map from ``cmd.lib.setup.variables``, or ``None`` if 

66 that file was not found. 

67 groups : dict[int, str] 

68 Group number -> output directory prefix, from ``cmds.start.*``. 

69 only : tuple[int, ...] 

70 Group numbers to keep; blocks in other groups go to :attr:`ignored`. 

71 blocks : list[dict] 

72 Decoded material records for the kept groups (populated by :meth:`decode`). 

73 ignored : list[dict] 

74 Decoded records whose group was not in :attr:`only`. 

75 """ 

76 def __init__( 

77 self, 

78 path: Union[str, Path], 

79 groups: Tuple[int, ...] = (1, 2), 

80 decode: bool = True, 

81 raise_casts: bool = True, 

82 ): 

83 """ 

84 Parameters 

85 ---------- 

86 path : str or Path 

87 Path to the Tetracorder expert system file (``cmd.lib.setup.<version>``). 

88 May also be a directory, in which case the first file matching 

89 ``cmd.lib.setup.t*`` inside it is used. 

90 groups : tuple[int, ...], optional 

91 Group numbers to decode. Blocks belonging to any other group are parsed 

92 but set aside in :attr:`ignored` rather than :attr:`blocks`. Defaults to 

93 ``(1, 2)`` (the group 1 / group 2 minerals of the EMIT L2B product). 

94 Pass an empty/falsy value to keep every group found in :attr:`groups`. 

95 decode : bool, default=True 

96 Whether to immediately decode the file upon initialization. If False, the 

97 file can be decoded later by calling :meth:`decode`. 

98 raise_casts : bool, default=True 

99 Whether to raise exceptions when type casting fails in :meth:`cast`. If 

100 False, failed casts return the original value instead of raising an error. 

101 """ 

102 self.file = Path(path) 

103 if self.file.is_dir(): 

104 self.file = next(self.file.glob("cmd.lib.setup.t*")) 

105 

106 # Retrieve external references first, if available 

107 self.root = self.file.parent 

108 

109 self.vars = {} 

110 if (f := self.root / "cmd.lib.setup.variables").exists(): 

111 self.vars = self.parse_variables(f) 

112 

113 self.groups = {} 

114 if files := list(self.root.glob(f"cmds.start.*")): 114 ↛ 122line 114 didn't jump to line 122 because the condition on line 114 was always true

115 self.groups = self.parse_group_paths(files[0]) 

116 

117 # Unused 

118 # self.nots = {} 

119 # if (f := self.root / "cmd.lib.setup.nots-ratios").exists(): 

120 # self.nots = self.parse_not_ratios(f) 

121 

122 self.only = groups 

123 if not groups: 

124 self.only = list(self.groups) 

125 

126 self._raise = raise_casts 

127 

128 if decode: 

129 self.decode()

131 def decode(self) -> List[dict]: 

132 """ 

133 Read the expert system file and parse every group block. 

134 

135 The file is read line by line into :attr:`lines`, split into blocks by 

136 :meth:`extract_blocks`, and each block is parsed by :meth:`parse_block`. 

137 Parsed blocks are sorted into :attr:`blocks` (group in :attr:`only`) or 

138 :attr:`ignored` (any other group). 

139 

140 Returns 

141 ------- 

142 list[dict] 

143 The kept blocks, i.e. :attr:`blocks`. 

144 """ 

145 self.lines = [] 

146 with self.file.open("r") as file: 

147 for line in file: 

148 line = line.strip() 

149 self.lines.append(line) 

150 

151 

152 self.blocks = [] 

153 self.ignored = [] 

154 for block in self.extract_blocks(self.lines): 

155 block = self.parse_block(block) 

156 if block["group"] in self.only: 

157 self.blocks.append(block) 

158 else: 

159 self.ignored.append(block) 

160 

161 return self.blocks

163 def parse_block(self, block: List[str]) -> dict: 

164 """ 

165 Parse a single group block into a material record. 

166 

167 Each block is a list of stripped, comment-free lines (as produced by 

168 :meth:`extract_blocks`). Recognized directives are pulled out into a ``dict``; 

169 any ``[NAME]`` references are substituted with their values from :attr:`vars`. 

170 

171 Parameters 

172 ---------- 

173 block : list[str] 

174 Lines of one group block from the expert system file. 

175 

176 Returns 

177 ------- 

178 dict 

179 A material record with any of the following keys, depending on which 

180 directives the block contains: 

181 

182 ``group`` : int 

183 Group number the material belongs to. 

184 ``use`` : str 

185 Whether the entry is active (``"yes"``/``"no"``). 

186 ``name`` : str 

187 Short material name (first token after ``TITLE=``). 

188 ``title`` : str 

189 Full material title. 

190 ``library`` : str 

191 Spectral library ID the reference record lives in (e.g. ``splib06``). 

192 ``record`` : int 

193 Record number of the reference spectrum within ``library``. 

194 ``filename`` : str 

195 Output base file name written by the mapper. 

196 ``path`` : str 

197 ``filename`` prefixed with its group output directory, when the group 

198 directory is known from :attr:`groups`. 

199 ``data_type_scaling`` : float 

200 Scaling applied to the 8-bit (``0-255``) output DN. 

201 ``features`` : list[dict] 

202 Spectral features. Each feature has ``feature_type`` (``DLw``/``MLw``/ 

203 ``OLw``), a 4-element ``continuum`` (left/right window endpoints), and 

204 one list per fit/depth constraint keyword found on the feature line 

205 (e.g. ``ct``, ``rcbblc<``). 

206 ``constituent_constraints`` : dict[str, list[float]] 

207 Per-block constraint keyword -> values (e.g. ``FD-FIT>``, ``FITALL>``). 

208 """ 

209 data = {} 

210 for i, line in enumerate(block): 

211 if line.startswith("group"): 

212 data["group"] = int(line.split()[1]) 

213 

214 elif line.startswith("use="): 

215 data["use"] = line.split(" ")[1] 

216 

217 elif line.startswith("ID="): 

218 data["id"] = line[3:].strip() 

219 

220 elif "=- TITLE=" in line: 

221 split = line.split("TITLE=")[1].split() 

222 data["name"] = split[0] 

223 data["title"] = " ".join(split) 

224 

225 elif line.startswith("define library records"): 

226 split = block[i+1].split() 

227 data["library"] = split[2].strip("[]") 

228 data["record"] = int(split[3]) 

229 

230 elif line.startswith("define output"): 

231 data["filename"] = block[i+2] 

232 data["path"] = None 

233 if path := self.groups.get(data["group"]): 

234 data["path"] = path + data["filename"] 

235 

236 elif line.startswith("8 DN 255"): 

237 val = line.split(" ")[-1] 

238 val = self.vars.get(val.strip("[]"), val) # Substitute variables, if necessary 

239 data["data_type_scaling"] = self.cast(float, val) 

240 

241 elif line.startswith("define features"): 

242 data["features"] = [] 

243 while (line := block[i]) != "endfeatures": 

244 i += 1 

245 # Substitute variables, if necessary 

246 split = [self.vars.get(v.strip("[]"), v) for v in line.split()] 

247 if line.startswith("f") and split[1] in ("DLw", "MLw", "OLw"): 

248 feat = { 

249 "feature_type": split[1], 

250 "continuum": self.cast(float, split[2:6]), 

251 } 

252 

253 line = " ".join(split[6:]) 

254 matches = re.findall(r"([^\s\d]+)>?\s+(.+?)(?=\s+[^\s\d]+>?[\s]|$)", line) 

255 for key, val in matches: 

256 feat[key] = self.cast(float, val.split()) 

257 

258 data["features"].append(feat) 

259 

260 elif line.startswith("constraint:"): 

261 consts = data.setdefault("constituent_constraints", {}) 

262 # Match successive KEY>VALUE pairs, where VALUE extends until the next KEY> or end of line 

263 matches = re.findall(r"([^\s>]+)>\s*(.*?)(?=\s+[^\s>]+>|$)", line) 

264 for key, val in matches: 

265 val = self.vars.get(val.strip("[]"), val) 

266 consts[key] = self.cast(float, val.split()) 

267 

268 return data

289 def cast(self, dtype: type, val: Any) -> Any: 

290 """ 

291 Cast a value (or each item of a list) to ``dtype``, tolerating failures. 

292 

293 Applies ``dtype`` to ``val`` (or elementwise if ``val`` is a list). When a 

294 conversion fails, the original value is kept unless ``raise_casts`` was set on 

295 the decoder, in which case the error propagates. Used to coerce the numeric 

296 fields parsed out of the expert system while leaving unparseable tokens (e.g. 

297 unresolved ``[NAME]`` references) intact. 

298 

299 Parameters 

300 ---------- 

301 dtype : type 

302 The target type/callable applied to each value (e.g. ``float``). 

303 val : Any 

304 A single value or a list of values to cast. 

305 

306 Returns 

307 ------- 

308 Any 

309 The cast value, or a list of cast values when ``val`` is a list. 

310 """ 

311 def apply(v: Any) -> Any: 

312 """Apply dtype conversion to a single value, tolerating failures.""" 

313 try: 

314 return dtype(v) 

315 except: 

316 if self._raise: 

317 raise 

318 return v 

319 

320 if isinstance(val, list): 

321 return [apply(v) for v in val] 

322 return apply(val)

324 def table(self, 

325 groups: Optional[Iterable[int]] = None, 

326 columns: Tuple[str, ...] = ("id", "group", "library", "record", "title", "path"), 

327 pandas: bool = True 

328 ) -> Union[pd.DataFrame, List[list]]: 

329 """ 

330 Collect the decoded records into a tabular form, one row per material. 

331 

332 Selects one row per material across both :attr:`blocks` and :attr:`ignored` 

333 whose group is in ``groups`` (so records outside :attr:`only` can be included 

334 by widening ``groups``), keeping only the fields named in ``columns``. 

335 

336 Parameters 

337 ---------- 

338 groups : Iterable[int], optional 

339 Group numbers to include. Defaults to :attr:`only` (the groups that were 

340 decoded into :attr:`blocks`). 

341 columns : tuple[str, ...], optional 

342 Record keys to emit as columns, in order. Every named key must be present 

343 on each included record (see :meth:`parse_block` for available keys). 

344 Defaults to ``("group", "library", "record", "title", "path")``. 

345 pandas : bool, default=True 

346 If True, return a :class:`pandas.DataFrame`. If False, return a list of 

347 rows with ``columns`` as the leading header row. 

348 

349 Returns 

350 ------- 

351 pandas.DataFrame or list[list] 

352 A DataFrame of the selected records when ``pandas`` is True; otherwise a 

353 list of rows whose first element is the ``columns`` header. 

354 """ 

355 if groups is None: 

356 groups = self.only 

357 

358 data = [columns] 

359 for block in self.blocks + self.ignored: 

360 if block["group"] in groups: 

361 data.append([block[c] for c in columns]) 

362 

363 if pandas: 

364 return pd.DataFrame(data[1:], columns=columns) 

365 return data

367 def match_ref( 

368 self, 

369 matrix: Union[str, Path], 

370 nu: Optional[pd.DataFrame] = None, 

371 keys: list[str] = ["id"], 

372 sortby: str = None, 

373 clean_titles: bool = False, 

374 ) -> pd.DataFrame: 

375 """ 

376 Align the decoded records to a reference matrix's stable ``index`` column. 

377 

378 Merges the ``index`` column from an existing reference matrix CSV onto the 

379 decoded table, matching on ``record`` / ``library`` so each material keeps the 

380 index it had in the reference. Records with no match in the reference are 

381 treated as new and assigned fresh indices continuing past the reference's max, 

382 and the additions are logged. 

383 

384 Parameters 

385 ---------- 

386 matrix : str or Path 

387 Path to the reference matrix CSV. It must contain ``record``, ``library``, 

388 and ``index`` columns (column names are matched case-insensitively). 

389 nu : pandas.DataFrame, optional 

390 The table to align. Defaults to :meth:`table` (the decoded records). Must 

391 contain ``record`` and ``library`` columns. 

392 keys : list[str], default=["record", "library", "group"] 

393 Column names to use as the merge key when matching records between the 

394 reference matrix and the new table. 

395 sortby : str, optional 

396 Column name to sort the output by. If None (default), the order is 

397 preserved from the input. 

398 clean_titles : bool, default=False 

399 Whether to remove family identifiers (substrings containing ``=``) from 

400 the ``title`` column. 

401 

402 Returns 

403 ------- 

404 pandas.DataFrame 

405 ``nu`` with a leading ``index`` column carrying the reference indices, and 

406 newly-assigned indices for records absent from the reference. 

407 """ 

408 og = pd.read_csv(matrix) 

409 

410 if nu is None: 

411 nu = self.table() 

412 

413 # Backwards compatibility that used Capitalized columns 

414 og = og.rename(columns={c: c.lower() for c in og}) 

415 

416 columns = keys + ["index"] 

417 if "url" in og: 417 ↛ 418line 417 didn't jump to line 418 because the condition on line 417 was never true

418 columns.append("url") 

419 

420 # Merge using record/library 

421 nu = nu.merge( 

422 og[columns], 

423 on=keys, 

424 how="left", 

425 ) 

426 nu["index"] = nu["index"].astype("Int64") 

427 

428 # Reorder index to the first column 

429 nu.insert(0, "index", nu.pop("index")) 

430 

431 # Update index of missing entries 

432 nul = nu["index"].isnull() 

433 

434 # Continue past the reference's max (not just the matched rows') so a new record 

435 # never reuses the index of one that was dropped; start at 1 if nothing matched 

436 top = og["index"].max() 

437 off = 1 if pd.isna(top) else int(top) + 1 

438 rng = range(off, off + nul.sum()) 

439 

440 if sortby: 

441 nu = nu.sort_values(by=sortby) 

442 

443 if clean_titles: 

444 # Remove "family" identifiers from title strings (if the substring has an `=`) 

445 nu["title"] = nu["title"].str.replace(r"\s*\S*=[^\s]*", "", regex=True) 

446 

447 new = nu[nul] 

448 fmt = new.set_index("index").to_string(index=False) 

449 Logger.info(f"{len(new)} records are new:\n{fmt}") 

450 Logger.info(f"Setting these as {rng}") 

451 

452 nu.loc[nul, "index"] = rng 

453 

454 return nu

509 def parse_variables(file: str) -> Dict[str, str]: 

510 """ 

511 Parse ==[NAME] value... definitions from a command file. 

512 

513 This function extracts variable definitions from tetracorder command files, 

514 which use a special ==[NAME] syntax to define parameters. The values are 

515 returned as strings to preserve formatting and precision. 

516 

517 Parameters 

518 ---------- 

519 file : str 

520 Path to the command file to parse. 

521 

522 Returns 

523 ------- 

524 Dict[str, str] 

525 Mapping of variable names to their string values (whitespace-stripped). 

526 If a variable is defined multiple times, the last definition wins. 

527 

528 Examples 

529 -------- 

530 Given a file with content: 

531 ==[THRESHOLD] 0.5 

532 ==[VALUES] 1.0 2.0 3.0 

533 ==[THRESHOLD] 0.8 

534 

535 >>> parse_variables("cmd.file") 

536 {'THRESHOLD': '0.8', 'VALUES': '1.0 2.0 3.0'} 

537 """ 

538 text = Path(file).read_text() 

539 

540 pattern = re.compile( 

541 r"^==\[(?P<name>[^\]]+)\]\s*(?P<values>[-+0-9.eE\s]+)", 

542 re.MULTILINE, 

543 ) 

544 

545 return { 

546 match["name"]: match["values"].strip() 

547 for match in pattern.finditer(text) 

548 }

609 @staticmethod 

610 def extract_blocks(lines: Iterable[str]) -> Iterator[List[str]]: 

611 """ 

612 Split the expert system lines into group blocks. 

613 

614 A block opens on a line matching ``group <N>`` and closes on the line 

615 starting with ``endaction``. Within a block, trailing ``\\#`` comments are 

616 stripped (except on ``TITLE`` lines, where ``#`` may be significant) and blank 

617 lines are dropped, so each yielded block is a list of meaningful lines only. 

618 

619 Parameters 

620 ---------- 

621 lines : Iterable[str] 

622 Stripped lines of the expert system file. 

623 

624 Yields 

625 ------ 

626 list[str] 

627 One group block, from its ``group <N>`` line through ``endaction``. 

628 """ 

629 block = None 

630 for line in lines: 

631 if re.match(r"^group\s+\d", line): 

632 block = [] 

633 

634 if block is not None: 

635 # Strip comments from the line 

636 if "TITLE" not in line: 

637 line = line.split("\\#")[0].strip() 

638 

639 if line: 

640 block.append(line) 

641 if line.startswith("endaction"):
